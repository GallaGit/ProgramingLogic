Fizz Buzz

Del 0 al 100: multiplos de 3 → Fizz, de 5 → Buzz, de ambos → Fizz Buzz.

El codigo

let i = 0;
for (i = 0; i <= 100; i++) {

    if (i % 3 === 0 && i % 5 === 0) {
        console.log("Fizz" + " Buzz");
    } else if (i % 3 === 0) {
        console.log("Fizz");
    } else if (i % 5 === 0) {
        console.log("Buzz");
    } else {
        console.log(i);
    }

}
